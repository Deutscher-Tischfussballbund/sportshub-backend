// sportshub-backend — current data model (league tree, docs/09-league-model.md)
// Season -> League -> Tier -> Group -> Round -> MatchDay -> Match -> Set/Event
// plus placement (TeamParticipation) / roster (RosterEntry), rule sets, and scoped RBAC.
// All ids are 14-char nano-ids (varchar). Every parent link is a child-side FK (@ManyToOne).
//
// Team versioning + entity history (docs/14-team-player-versioning.md): `team` is season-scoped,
// copied fresh into every season by CopyForwardService (same pattern as league/tier/comp_group),
// so a rename never rewrites a past season's display; it links its season-copies via
// `team_identity_id` (a new synthetic field). `player` and `club` are NOT season-scoped -- each
// stays a single row, and a rename is instead recorded in `entity_history` (old/new value +
// timestamp per field), reconstructible "as of" a point in time. `app_user` is the permanent,
// season-independent Keycloak login identity split out of the old `player` table —
// `role_assignment` now points there instead: a role belongs to a login identity, not the
// competitor record itself. `club_membership` is a first-class player<->club relationship,
// independent of team rosters: a player must be an active member of a club (joined_at set,
// left_at null) before RosterService#addPlayer will roster them onto one of that club's teams.
//
// Relationships are declared as standalone `Ref:` statements at the bottom, not inline on the
// column, since almost every FK here is nullable (a team can exist before being placed, a
// season can be archived without every child set, etc.) — dbdiagram.io's inline
// `[ref: > ...]` column attribute requires the column to be NOT NULL, which most of these
// genuinely aren't.
//
// The few FKs actually declared `@ManyToOne(optional = false)` in code use plain `>` below:
// role_assignment.user_id and standing.group_id/team_id.
//
// Machine access (docs/20-api-keys.md): `api_key` holds backend-issued, read-only API keys (only the
// SHA-256 hash is stored) -- independent of `role_assignment`, no FK; writes always need a human
// login. `release_note` is unrelated app content (the "what's new" feed), standalone with no
// relationships at all.

// ============================ Enums ============================

Enum GroupState {
  PLANNED
  READY
  RUNNING
  FINISHED
  CANCELED
}

Enum ResultState {
  OPEN      [note: 'No result entered yet']
  SUBMITTED [note: 'Entered by a team member; waiting for both captains to agree (home/away_confirmed_at)']
  CONFIRMED [note: 'Both captains agreed, or a neutral admin entered/confirmed -> standings recomputed']
}

Enum MatchState {
  OPEN       [note: 'Waiting to be announced']
  PAUSED     [note: 'Disabled (on pause)']
  SKIPPED    [note: 'Skipped (e.g. entry left tournament)']
  RUNNING    [note: 'Running']
  PLAYED     [note: 'Finished / played']
  PLANNED    [note: 'Planned / awaiting announcement']
  INCOMPLETE [note: 'Waiting for entries to be assigned']
  BYE        [note: 'A bye (e.g. in elimination stage)']
}

Enum MatchType {
  SINGLE
  DOUBLE
  GOALIE
}

Enum Winner {
  HOME
  AWAY
  DRAW
}

Enum MatchEventType {
  GOAL
  OWN_GOAL
  TIMEOUT
  START
  END
  CARD
  OTHER
}

Enum RosterStatus {
  DRAFT     [note: 'Team admin edits the roster']
  SUBMITTED [note: 'Locked by the team; awaiting admin confirmation']
  CONFIRMED [note: 'Confirmed by an admin above the team']
}

Enum PlaySystem {
  ROUND_ROBIN
  SWISS [note: 'Retained; most common mode in table soccer, primarily a (parked) tournament format']
}

Enum ParticipationStatus {
  ACTIVE
  WITHDRAWN [note: 'Team dropped out of the league; roster locked, excluded from copy-forward']
}

Enum MatchdayDecision {
  ALL_GAMES [note: 'Decided once all planned games are played']
  FIRST_TO  [note: 'Decided once a team reaches matchday_target game wins']
}

Enum SchedulingMode {
  DAY_BATCH [note: 'The organizer sets the dates: in batches per day, or as fixed slots at generation (tournament weekend). Teams cannot propose.']
  WINDOW    [note: 'Each round gets a date window; the two teams of a fixture agree within it (only mode with propose/accept)']
}

Enum SchedulingState {
  DEFAULT   [note: 'start_date/location are just the generated default']
  PROPOSED  [note: 'One side has proposed a real date/location']
  CONFIRMED [note: 'Final: both sides agreed (WINDOW), an admin set it, or a fixed slot at generation']
}

Enum Role {
  ADMIN
  REGION_ADMIN
  CLUB_ADMIN
  TEAM_ADMIN
  LEAGUE_ADMIN
  TOURNAMENT_UPLOADER        [note: 'Deprecated']
  REGION_TOURNAMENT_UPLOADER [note: 'Deprecated']
}

Enum ScopeType {
  GLOBAL
  REGION
  CLUB
  TEAM
  LEAGUE [note: 'scope_id is a league id']
}

// ==================== Identity & access ====================

Table federation {
  id varchar(14) [pk, note: 'A Landesverband / region, or the root federation (DTFB)']
  name varchar
  default_rule_set_id varchar [note: 'The template (blueprint) a new league copies when none is chosen (docs/21); null = the seeded DTFB Standard. Not read at runtime']
  parent_federation_id varchar [note: 'self-referencing tree; null = the single root federation (docs/16)']
}

Table club {
  id varchar(14) [pk, note: 'A single row per club, not season-scoped; a rename is tracked in entity_history']
  name varchar [not null]
  short_name varchar
  city varchar
  active boolean [not null, default: true]
  federation_id varchar [not null, note: 'plain FK to federation (no @ManyToOne)']
}

Table team {
  id varchar(14) [pk, note: 'One season-copy of a team; a fresh row per season via CopyForwardService']
  season_id varchar [note: 'Nullable: a team can exist before being placed into any season']
  name varchar
  club_id varchar
  team_identity_id varchar [not null, note: 'Stable across every season-copy of this team; carried forward verbatim by copy-forward']
  copied_from_team_id varchar [note: 'Audit pointer to the row this was cloned from, not a FK']
}

Table app_user {
  id varchar(14) [pk, note: 'The permanent, season-independent Keycloak login identity']
  dtfb_id varchar [unique, not null, note: 'Keycloak username -> dtfb_id']
  email varchar
  first_name varchar
  last_name varchar
}

Table player {
  id varchar(14) [pk, note: 'A single row per competitor, not season-scoped; a rename is tracked in entity_history']
  user_id varchar [note: 'Nullable: a captain-entered athlete may never log in themselves']
  first_name varchar [not null]
  last_name varchar [not null]
  national_id varchar
  international_id varchar
  nationality varchar
  birth_year int [not null]
  gender varchar [not null, note: 'PlayerGender: MALE | FEMALE | DIVERSE_MEN | DIVERSE_WOMEN; divers league side is admin-only']
  national_license varchar
  active boolean [not null, default: true]
}

Table club_membership {
  id varchar(14) [pk, note: 'A player belonging to a club -- the precondition for rostering them onto one of the club team rows via RosterService#addPlayer']
  player_id varchar
  club_id varchar
  joined_at datetime [not null]
  left_at datetime [note: 'Null = active; set = soft-removed (left the club)']
}

Table entity_history {
  id varchar(14) [pk]
  entity_type varchar [note: 'PLAYER | CLUB -- which table entity_id refers to']
  entity_id varchar [note: 'plain id, not a FK -- history survives even if the entity is later deleted']
  field_name varchar
  old_value text
  new_value text
  changed_at datetime [not null]
  changed_by_dtfb_id varchar

  Indexes {
    (entity_type, entity_id)
  }
}

Table category {
  id varchar(14) [pk, note: 'Global classification: Herren / Damen / Open']
  name varchar [not null]
  short_name varchar [not null, unique, note: 'Unique case-insensitively (utf8mb4_0900_ai_ci)']
  eligible_side varchar [null, note: 'LeagueSide MEN / WOMEN; NULL = open to every gender. Checked on roster add + submit (docs/19)']
}

Table location {
  id varchar(14) [pk]
  name varchar
  address varchar
  federation_id varchar [note: 'Owning region, fixed after creation; null = global/nationwide venue (API only, open decision)']
}

Table role_assignment {
  id varchar(14) [pk]
  user_id varchar [not null, note: 'A role belongs to the login identity, not a season-scoped Player row']
  role Role [not null]
  scope_type ScopeType [not null]
  scope_id varchar [note: 'id of the scoped entity (null for GLOBAL); for TEAM this is team_identity_id and for LEAGUE league_identity_id (not row ids), so a grant survives copy-forward; for CLUB it is the plain club row id (Club is not season-scoped)']
  granted_by_dtfb_id varchar
  created_at datetime [not null]
}

Table api_key {
  id varchar(14) [pk, note: 'Backend-issued, read-only API key for a machine consumer (docs/20); sent as X-API-Key; GET/HEAD only, never /v1/admin/** or /v1/auth/**']
  name varchar [not null, note: 'Human label for the admin UI, e.g. "Public Results Site"']
  key_prefix varchar [not null, note: 'First characters of the key, shown so keys can be told apart']
  key_hash varchar [unique, not null, note: 'Hex SHA-256 of the key; the plaintext is only returned once, at creation']
  active boolean [not null, default: true]
  expires_at date [null, note: 'Last valid day (inclusive); NULL = never expires']
  created_at timestamp [not null]
  created_by_dtfb_id varchar
  last_used_at timestamp [note: 'Refreshed at most once a minute']
}

// ==================== League tree ====================

Table season {
  id varchar(14) [pk, note: 'League time container']
  name varchar
  federation_id varchar
  start_date date
  end_date date
  registration_opens_at date [note: 'Required for registration to ever be open']
  registration_closes_at date [note: 'null = stays open indefinitely once opened']
  archived_at datetime [note: 'Soft delete']
}

Table league {
  id varchar(14) [pk, note: 'One category ladder in a season (was Competition)']
  league_identity_id varchar [not null, note: 'Stable across the league season-copies (like team.team_identity_id): generated once, carried by copy-forward; LEAGUE_ADMIN grants point to it (SPO-28, docs/14). V14 backfilled it from each row id']
  season_id varchar
  category_id varchar
  rule_set_id varchar [note: 'The league own rules: a snapshot private to this league (docs/21), copied from a template at create/copy-forward, frozen once the season has ended']
  name varchar
  import_id varchar
}

Table tier {
  id varchar(14) [pk, note: 'Promotion/relegation level, e.g. 1./2. Bayernliga']
  league_id varchar
  rule_set_id varchar [note: 'Optional override: a snapshot private to this tier (docs/21); null = the league rules apply']
  name varchar
  level int [note: 'Ordinal ladder position, 1 = top tier; orders promote/relegate']
}

Table comp_group {
  id varchar(14) [pk, note: 'The Group entity (one round-robin table); table name comp_group']
  tier_id varchar
  name varchar
  group_state GroupState [not null]
}

Table round {
  id varchar(14) [pk, note: 'A matchday number (Spieltag)']
  group_id varchar
  name varchar
  round_index int [note: 'Java/API field is "index"; column renamed -- "index" is a reserved MySQL word']
  window_start datetime [note: 'Only set under LeagueRuleSet.scheduling_mode = WINDOW (docs/12-matchday-scheduling.md)']
  window_end datetime
}

Table match_day {
  id varchar(14) [pk, note: 'A team-vs-team fixture']
  round_id varchar
  location_id varchar
  team_home_id varchar
  team_away_id varchar
  name varchar
  start_date datetime [not null]
  end_date datetime
  result_state ResultState [not null, default: 'OPEN']
  submitted_by_dtfb_id varchar
  home_confirmed_at datetime
  away_confirmed_at datetime
  bye boolean [not null, default: false, note: 'Fixture against the bye: team_away_id NULL, scored with the bye score (docs/22)']
  decided_at datetime [note: 'When the result became decided -- start of the confirmation deadline']
  // Scheduling lifecycle for start_date/location -- independent of result_state above; see
  // docs/12-matchday-scheduling.md.
  scheduling_state SchedulingState [not null, default: 'DEFAULT']
  schedule_proposed_by_dtfb_id varchar
  schedule_confirmed_at datetime
}

// ==================== Match detail ====================

Table match_game {
  id varchar(14) [pk, note: 'An individual game within a match day; JPA entity is Match, table is match_game -- "match" is a reserved MySQL word (collides with MATCH()...AGAINST())']
  match_day_id varchar
  position int [note: '1-based order within the match day; maps to game_plan_entry.position']
  start_time datetime [not null]
  end_time datetime
  home_score int
  away_score int
  state MatchState
  type MatchType
  winner Winner
}

Table match_set {
  id varchar(14) [pk]
  match_id varchar
  set_number int
  home_score int
  away_score int
}

Table match_event {
  id varchar(14) [pk]
  match_id varchar
  team_id varchar
  player_id varchar [note: 'plain id, not a FK']
  timestamp datetime
  type MatchEventType
  home_score int
  away_score int
  json text [note: '@Lob free-form payload']
}

Table standing {
  id varchar(14) [pk, note: 'A league-table row; unique (group_id, team_id)']
  group_id varchar [not null]
  team_id varchar [not null]
  played int
  wins int
  draws int
  losses int
  points int [note: 'Awarded from the effective LeagueRuleSet']
  sets_won int
  sets_lost int

  Indexes {
    (group_id, team_id) [unique]
  }
}

// ==================== Registration (placement + roster) ====================

Table team_participation {
  id varchar(14) [pk, note: 'A team placed in a league (season derived from league)']
  team_id varchar
  league_id varchar
  group_id varchar [note: 'null = registered but unplaced']
  copied_from_participation_id varchar [note: 'copy-forward / promotion-relegation chain']
  roster_status RosterStatus [not null, default: 'DRAFT']
  status ParticipationStatus [not null, default: 'ACTIVE']
  withdrawn_at datetime [note: 'Set when status becomes WITHDRAWN']
}

Table roster_entry {
  id varchar(14) [pk]
  participation_id varchar
  player_id varchar
  added_at datetime [not null]
  removed_at datetime [note: 'null = active; set = soft-removed']
}

// ==================== Rule sets ====================

Table league_rule_set {
  id varchar(14) [pk, note: 'Typed league rules in one of two roles (docs/21): a template (blueprint) in a federation library, or a snapshot owned by exactly one league/tier, frozen once its season has ended (409 RULE_SET_FROZEN)']
  federation_id varchar [note: 'null = DTFB-global template; a snapshot carries its season federation']
  snapshot boolean [not null, note: 'true = private rules of one league/tier; false = template']
  archived boolean [not null, note: 'Templates only: hidden from pickers and the main list']
  source_blueprint_id varchar [note: 'Snapshots only: the template it was copied from (null once that is deleted)']
  name varchar [note: 'Templates: unique per federation (or DTFB-wide), case-insensitive; snapshots may share names']
  play_system PlaySystem
  points_win int
  points_draw int [note: 'null = draws impossible']
  points_loss int
  sets_per_game int [note: 'best-of-N sets in one match']
  points_to_win_set int
  matchday_decision MatchdayDecision
  matchday_target int [note: 'the N for FIRST_TO']
  side_switch_allowed boolean
  min_roster_size int [note: 'Enforced by RosterService on submit; null = unconstrained']
  max_roster_size int [note: 'Enforced by RosterService on addPlayer; null = unconstrained']
  scheduling_mode SchedulingMode [note: 'Read by FixtureGenerationService; null = no convention configured yet (generation requires picking one)']
  scheduling_window_days int [note: 'Only meaningful when scheduling_mode = WINDOW']
  fixture_mode varchar [note: 'Rule profile RACE | GAMES (docs/22); NULL behaves as GAMES']
  race_target int [note: 'RACE: e.g. 42']
  race_step int [note: 'RACE: points per segment, e.g. 6']
  race_end_rule varchar [note: 'RACE: DRAW_ALLOWED (41:41) | TWO_POINT_LEAD']
  race_bye_score_winner int [note: 'RACE: bye fixture score, e.g. 42']
  race_bye_score_loser int [note: 'RACE: e.g. 30']
  confirmation_minutes int [note: 'Time to confirm once decided; NULL = no deadline']
}

Table game_plan_entry {
  id varchar(14) [pk, note: 'One ordered slot in a rule set matchday game plan']
  rule_set_id varchar
  position int
  game_type MatchType
}

// ==================== Content ====================

Table release_note {
  id varchar(14) [pk, note: 'Functional "what\'s new" feed shown to every logged-in user (shell bell button) -- global-admin authored. Distinct from technical/dev release notes, which are GitHub\'s own auto-generated notes on tagged releases']
  title_de varchar [not null]
  title_en varchar [not null]
  body_de varchar [not null]
  body_en varchar [not null]
  published_at datetime [not null, note: 'Set on create, preserved on edit']
}

// ==================== Tester tracker (separate subsystem) ====================
// Real tester feedback, unlike the disposable demo/seed domain data; planned to move into its own
// MySQL schema at the next wipe-requiring migration. No FK into the domain tables.

Table tracker_issue {
  id varchar(14) [pk]
  title varchar(200) [not null]
  description varchar(4000)
  status varchar [not null, note: 'TrackerIssueStatus: OPEN | APPROVED (converted to a GitHub issue) | DONE; varchar since V6']
  reported_by varchar
  github_repo varchar [note: 'Set when converted to a GitHub issue']
  github_issue_url varchar
  created_at timestamp [not null]
}

Table tracker_issue_vote {
  id varchar(14) [pk]
  issue_id varchar [not null]
  voter_name varchar [not null, note: 'Free text from the public tracker page; unique per issue']

  indexes {
    (issue_id, voter_name) [unique]
  }
}

// ============================ Relationships ============================
// Declared standalone (not inline on the column) for readability at this scale. Almost every FK
// here is nullable at the JPA/DB level (plain @ManyToOne, not `optional = false`) even where the
// service layer enforces "required" at the application level -- so the relationship operator
// uses the `?` optional modifier (`>?`, optional side = the column/left side) wherever the
// column itself is nullable.

Ref: federation.default_rule_set_id >? league_rule_set.id
Ref: federation.parent_federation_id >? federation.id
// club.federation_id is deliberately NOT drawn as a relationship: it's a plain String field in
// the JPA entity (no @ManyToOne), per the column note above.
Ref: team.season_id >? season.id
Ref: team.club_id >? club.id
Ref: player.user_id >? app_user.id
Ref: club_membership.player_id >? player.id
Ref: club_membership.club_id >? club.id
Ref: location.federation_id >? federation.id
Ref: role_assignment.user_id > app_user.id
Ref: season.federation_id >? federation.id
Ref: league.season_id >? season.id
Ref: league.category_id >? category.id
Ref: league.rule_set_id >? league_rule_set.id
Ref: tier.league_id >? league.id
Ref: tier.rule_set_id >? league_rule_set.id
Ref: comp_group.tier_id >? tier.id
Ref: round.group_id >? comp_group.id
Ref: match_day.round_id >? round.id
Ref: match_day.location_id >? location.id
Ref: match_day.team_home_id >? team.id
Ref: match_day.team_away_id >? team.id
Ref: match_game.match_day_id >? match_day.id
Ref: match_set.match_id >? match_game.id
Ref: match_event.match_id >? match_game.id
Ref: match_event.team_id >? team.id
Ref: standing.group_id > comp_group.id
Ref: standing.team_id > team.id
Ref: team_participation.team_id >? team.id
Ref: team_participation.league_id >? league.id
Ref: team_participation.group_id >? comp_group.id
Ref: roster_entry.participation_id >? team_participation.id
Ref: roster_entry.player_id >? player.id
Ref: game_plan_entry.rule_set_id >? league_rule_set.id
Ref: league_rule_set.source_blueprint_id >? league_rule_set.id
Ref: tracker_issue_vote.issue_id > tracker_issue.id
